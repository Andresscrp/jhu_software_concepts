"""Module 9 — KMeans clustering and analysis of GradCafe program data."""
from __future__ import annotations

from typing import Any

import mlflow
import mlflow.sklearn
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.cluster import KMeans
from sklearn.decomposition import PCA
from sklearn.feature_extraction.text import TfidfVectorizer


def looks_like_university(value: str) -> bool:
    """Return True if text looks like a school or university name."""
    lowered = value.strip().lower()

    school_keywords = (
        "university",
        "college",
        "institute",
        "school",
        "polytechnic",
        "state",
        "campus",
    )

    return any(keyword in lowered for keyword in school_keywords)


def load_and_prepare_data(filepath: str) -> pd.DataFrame:
    """Load the GradCafe dataset and prepare Program and University columns."""
    df = pd.read_json(filepath)

    df = df[df["program"].notna()].copy()

    df["program"] = (
        df["program"]
        .astype(str)
        .str.strip()
        .str.replace(r"\s+", " ", regex=True)
    )

    df["Program"] = df["program"]

    university_candidates = (
        "school",
        "university",
        "institution",
        "college",
        "school_name",
        "institution_name",
    )

    chosen_university_column = None
    for column_name in university_candidates:
        if column_name in df.columns and df[column_name].notna().any():
            chosen_university_column = column_name
            break

    if chosen_university_column is not None:
        df["University"] = (
            df[chosen_university_column]
            .fillna("Unknown")
            .astype(str)
            .str.strip()
            .str.replace(r"\s+", " ", regex=True)
        )
        df.loc[df["University"] == "", "University"] = "Unknown"
    else:
        df["University"] = "Unknown"

    split_cols = df["program"].str.split(",", n=1, expand=True)

    if split_cols.shape[1] > 1:
        first_part = split_cols[0].fillna("").astype(str).str.strip()
        second_part = split_cols[1].fillna("").astype(str).str.strip()

        should_split = second_part.apply(looks_like_university)
        university_unknown = df["University"] == "Unknown"

        df.loc[should_split, "Program"] = first_part.loc[should_split].values
        df.loc[
            should_split & university_unknown,
            "University",
        ] = second_part.loc[should_split & university_unknown].values

    print(f"Number of Entries: {len(df)}")
    print(f"Number of Unique Program Input Names: {df['Program'].nunique()}")

    return df


def vectorize_programs(df: pd.DataFrame) -> tuple[TfidfVectorizer, Any]:
    """Convert program names into TF-IDF vectors."""
    vectorizer = TfidfVectorizer(stop_words="english")
    tfidf_matrix = vectorizer.fit_transform(df["Program"])

    print("TF-IDF matrix shape:", tfidf_matrix.shape)
    print("TF-IDF matrix type:", type(tfidf_matrix))
    print("Nonzero entries:", tfidf_matrix.nnz)

    return vectorizer, tfidf_matrix


def run_initial_clustering(tfidf_matrix: Any) -> tuple[pd.Series, Any]:
    """Reduce TF-IDF matrix to 2D and run initial KMeans clustering."""
    pca_2d = PCA(n_components=2, random_state=42)
    reduced_data = pca_2d.fit_transform(tfidf_matrix.toarray())

    # MLflow tracking setup
    mlflow.set_tracking_uri("http://localhost:8080")
    mlflow.set_experiment("KMeans Clustering")

    params = {
        "max_iter": 500,
        "n_clusters": 25,
        "n_init": 5,
        "random_state": 42,
    }

    with mlflow.start_run():

        mlflow.log_params(params)

        kmeans = KMeans(**params)
        cluster_labels = kmeans.fit_predict(reduced_data)

        mlflow.log_metric("inertia", kmeans.inertia_)

        mlflow.sklearn.log_model(kmeans, "model")

        mlflow.register_model(
            f"runs:/{mlflow.active_run().info.run_id}/model",
            "Clustering"
        )

    plt.figure(figsize=(12, 9))
    scatter = plt.scatter(
        reduced_data[:, 0],
        reduced_data[:, 1],
        c=cluster_labels,
        cmap="tab20",
        s=20,
        alpha=0.8,
    )

    legend_handles, legend_labels = scatter.legend_elements()
    plt.legend(
        legend_handles,
        legend_labels,
        title="Clusters",
        loc="upper right",
        fontsize="small",
        title_fontsize="small",
    )

    plt.title("Kmeans Clustering of Programs")
    plt.xlabel("KMeans Distance Direction 1")
    plt.ylabel("KMeans Distance Direction 2")
    plt.tight_layout()
    plt.savefig("initial_cluster.png")
    plt.close()

    return pd.Series(cluster_labels), reduced_data


def attach_clusters(df: pd.DataFrame, cluster_labels: pd.Series) -> pd.DataFrame:
    """Attach cluster labels back to the dataframe and print preview."""
    df = df.copy()
    df["cluster"] = cluster_labels.values

    print("\nClustered DataFrame preview:")
    print(df[["Program", "University", "cluster"]].head(100).to_string())

    return df


def save_dataframe_image(df: pd.DataFrame) -> None:
    """Save the first 100 rows of the clustered dataframe as an image."""
    preview_df = df[["Program", "University", "cluster"]].head(100).copy()

    fig, ax = plt.subplots(figsize=(18, 30))
    ax.axis("off")

    table = ax.table(
        cellText=preview_df.values,
        colLabels=preview_df.columns,
        loc="center",
        cellLoc="left",
    )

    table.auto_set_font_size(False)
    table.set_fontsize(8)
    table.scale(1.0, 1.25)

    plt.tight_layout()
    plt.savefig("clustered_dataFrame.png", dpi=300, bbox_inches="tight")
    plt.close()


def make_elbow_plot(tfidf_matrix: Any) -> None:
    """Create elbow plot using PCA-reduced TF-IDF features."""
    pca_many = PCA(n_components=75, random_state=42)
    reduced_data = pca_many.fit_transform(tfidf_matrix.toarray())

    inertias = []
    k_values = range(1, 101)

    for k_value in k_values:
        kmeans = KMeans(
            n_clusters=k_value,
            max_iter=100,
            n_init=5,
            random_state=42,
        )
        kmeans.fit(reduced_data)
        inertias.append(kmeans.inertia_)

    plt.figure(figsize=(10, 6))
    plt.plot(k_values, inertias, marker="x", label="Inertia by K")
    plt.title("The Elbow Method using Inertia")
    plt.xlabel("Values of K")
    plt.ylabel("Inertia")
    plt.legend()
    plt.tight_layout()
    plt.savefig("elbow.png")
    plt.close()

    print("\nElbow plot saved as elbow.png")


def make_major_plots(df: pd.DataFrame) -> None:
    """Create GRE and GRE V boxplots for Philosophy and Computer Science clusters."""
    df = df.copy()

    df["GRE"] = pd.to_numeric(df.get("gre_general"), errors="coerce")
    df["GRE V"] = pd.to_numeric(df.get("gre_verbal"), errors="coerce")

    df = df[
        (df["GRE"].notna())
        & (df["GRE V"].notna())
        & (df["GRE"].between(130, 170))
        & (df["GRE V"].between(130, 170))
    ].copy()

    philosophy_rows = df[df["Program"].str.contains("Philosophy", case=False, na=False)]
    philosophy_cluster = philosophy_rows["cluster"].mode().iloc[0]
    philosophy_df = df[df["cluster"] == philosophy_cluster].copy()

    plt.figure(figsize=(8, 6))
    plt.boxplot([philosophy_df["GRE"], philosophy_df["GRE V"]], tick_labels=["GRE", "GRE V"])
    plt.plot([], [], label="Philosophy Cluster")
    plt.legend()
    plt.savefig("philosophy.png")
    plt.close()

    cs_rows = df[df["Program"].str.contains("Computer Science", case=False, na=False)]
    cs_cluster = cs_rows["cluster"].mode().iloc[0]
    cs_df = df[df["cluster"] == cs_cluster].copy()

    plt.figure(figsize=(8, 6))
    plt.boxplot([cs_df["GRE"], cs_df["GRE V"]], tick_labels=["GRE", "GRE V"])
    plt.plot([], [], label="Computer Science Cluster")
    plt.legend()
    plt.savefig("computer_science.png")
    plt.close()


def main() -> None:
    df = load_and_prepare_data("../module_8/cleaned_gradcafe.json")

    _, tfidf_matrix = vectorize_programs(df)
    cluster_labels, _ = run_initial_clustering(tfidf_matrix)

    clustered_df = attach_clusters(df, cluster_labels)

    save_dataframe_image(clustered_df)
    make_elbow_plot(tfidf_matrix)
    make_major_plots(clustered_df)


if __name__ == "__main__":
    main()
